# Control for e28, with no remote operation from L during the busy
# period. S has a second, idle UI thread. S's main thread is busy 1000 ms (no
# operation of L's waits for it); B, holding the
# second thread's window's element, calls ropname on it.
spawn S server
wait S hwnd
send S newthread
wait S hwnd

spawn L client $HWND L
send L init
wait L done
send L efh
wait L done
send L rop
wait L done 10000

spawn B client $HWND2 B
send B init
wait B done
send B efh
wait B done
send B ropname
wait B done 10000

send S busy 1000
wait S busy-begin
sleep 5
send B ropname
wait B done 15000
drain 300

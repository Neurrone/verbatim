# B's first MSAA call into S (msaa; UIA's MSAA bridge to S's provider)
# while L's remote operation into S waits for S's busy UI thread.
spawn S server
wait S hwnd

spawn L client $HWND L
send L init
wait L done
send L efh
wait L done
send L rop
wait L done 10000

spawn B client $HWND B
send B init
wait B done

send S busy 40
wait S busy-begin
send L async rop
sleep 5
send B msaa
wait B done 15000
wait L done 15000
drain 300
